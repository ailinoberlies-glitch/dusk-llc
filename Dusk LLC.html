OUR PHILOSOPHY

Where the city slows down.

Dusk Society was conceived in the gap between Baltimore's ambition and its evening. A city that moves — that builds, trades, creates — rarely gives itself permission to simply be.




We are not a restaurant. We are not a club. We are an argument for the long table — for the kind of conversation that requires a second bottle and no schedule for the morning.

Membership is by introduction. The calendar is curated. The door is always the same — heavy, dark, copper-handled — and it opens into something that feels, briefly, like nowhere else.

  function scrollTo(id) {

    document.getElementById(id).scrollIntoView({ behavior: 'smooth' });

  }


  function openTab(name, btn) {

    document.querySelectorAll('.tab-panel').forEach(p => p.classList.remove('active'));

    document.querySelectorAll('.tab-btn').forEach(b => b.classList.remove('active'));

    document.getElementById('tab-' + name).classList.add('active');

    btn.classList.add('active');

  }


  function triggerUpload(cell) {

    cell.querySelector('.file-input-hidden').click();

  }


  function loadImage(input) {

    const cell = input.closest('.mood-cell');

    const file = input.files[0];

    if (!file) return;

    const reader = new FileReader();

    reader.onload = e => {

      const placeholder = cell.querySelector('.mood-cell-placeholder');

      if (placeholder) placeholder.remove();

      let img = cell.querySelector('.mood-cell-img');

      if (!img) {

        img = document.createElement('img');

        img.className = 'mood-cell-img';

        cell.insertBefore(img, cell.querySelector('.mood-cell-overlay'));

      }

      img.src = e.target.result;

    };

    reader.readAsDataURL(file);

  }


  function openModal() {

    document.getElementById('modal').classList.add('open');

  }


  function closeModal() {

    document.getElementById('modal').classList.remove('open');

  }


  function closeModalBg(e) {

    if (e.target === document.getElementById('modal')) closeModal();

  }


  function submitModal() {

    closeModal();

    const toast = document.getElementById('toast');

    toast.classList.add('show');

    setTimeout(() => toast.classList.remove('show'), 3500);

  }


  // Scroll reveal

  const observer = new IntersectionObserver(entries => {

    entries.forEach(e => {

      if (e.isIntersecting) { e.target.classList.add('visible'); observer.unobserve(e.target); }

    });

  }, { threshold: 0.12 });


  document.querySelectorAll('.reveal').forEach(el => observer.observe(el));

</script>
